## Proposed Style Guide

**Font:** Helvetica primary (SVG) / Arial → Helvetica → DejaVu Sans fallback (matplotlib)

**Three tiers:**

| Tier | pt value | Used for |
|---|---|---|
| TITLE | **20** | Panel titles (`ax.set_title`), row/group headers |
| LABEL | **16** | Axis labels (`set_xlabel`/`set_ylabel`), legend frame titles |
| SMALL | **13** | Tick labels, legend text, captions, scale bar text, minor annotations |

**Canonical `rcParams` block:**
```python
{
    "font.family": "sans-serif",
    "font.sans-serif": ["Arial", "Helvetica", "DejaVu Sans"],
    "font.size": 13,           # base — covers anything not explicitly set
    "axes.titlesize": 20,
    "axes.labelsize": 16,
    "xtick.labelsize": 13,
    "ytick.labelsize": 13,
    "legend.fontsize": 13,
    "axes.linewidth": 0.8,
    "pdf.fonttype": 42,
    "ps.fonttype": 42,
    "svg.fonttype": "none",
    "savefig.dpi": 300,
}
```

**Inkscape SVG 3-tier targets** (anchored to the largest existing size in each document):

| File | TITLE (px) | LABEL (px) | SMALL (px) |
|---|---|---|---|
| fig1/fig1.svg | 2.9 | 2.32 | 1.885 |
| fig2/fig2.svg | 2.82222 | 2.258 | 1.834 |
| fig3/fig3.svg | 2.50329 | 2.003 | 1.627 |
| fig4/fig4.svg | 4.23333 | 3.387 | 2.752 |
| fig5/fig5.svg | 4.23333 | 3.387 | 2.752 |

Mapping rationale: scale down from largest existing size using the 20:16:13 ratios.  
Middle tier (LABEL) = largest × (16/20).  Small tier = largest × (13/20).

```python
"""Normalize font sizes in Inkscape assembly SVGs to 3-tier style guide values.

Each SVG has its own TITLE size (the largest existing size), from which LABEL
and SMALL are derived using the 20:16:13 ratios. Sizes within ±5% of a tier
boundary are snapped to that tier.

Run: uv run python docs/tex_draft/_normalize_svg_fonts.py --dry-run
     uv run python docs/tex_draft/_normalize_svg_fonts.py
"""
from __future__ import annotations
import argparse
import re
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[2]
FIGS = REPO_ROOT / "docs" / "tex_draft" / "figures"

# (TITLE_px, LABEL_px, SMALL_px) for each SVG
SVG_TIERS: dict[str, tuple[float, float, float]] = {
    "fig1/fig1.svg":  (2.9,      2.9 * 16/20, 2.9 * 13/20),
    "fig2/fig2.svg":  (2.82222,  2.82222 * 16/20, 2.82222 * 13/20),
    "fig3/fig3.svg":  (2.50329,  2.50329 * 16/20, 2.50329 * 13/20),
    "fig4/fig4.svg":  (4.23333,  4.23333 * 16/20, 4.23333 * 13/20),
    "fig5/fig5.svg":  (4.23333,  4.23333 * 16/20, 4.23333 * 13/20),
}

FONT_SIZE_PAT = re.compile(r'font-size:(\d+\.?\d*)px')


def nearest_tier(size: float, title: float, label: float, small: float) -> float:
    candidates = [title, label, small]
    return min(candidates, key=lambda t: abs(size - t))


def normalize_svg(path: Path, tiers: tuple[float, float, float], dry_run: bool) -> int:
    title, label, small = tiers
    text = path.read_text(encoding="utf-8")
    changes = 0

    def replace(m: re.Match) -> str:
        nonlocal changes
        original = float(m.group(1))
        snapped = nearest_tier(original, title, label, small)
        if abs(snapped - original) > 1e-6:
            changes += 1
            return f"font-size:{snapped:.5g}px"
        return m.group(0)

    new_text = FONT_SIZE_PAT.sub(replace, text)
    if changes and not dry_run:
        path.write_text(new_text, encoding="utf-8")
    return changes


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--dry-run", action="store_true")
    args = parser.parse_args()

    for rel, tiers in SVG_TIERS.items():
        path = FIGS / rel
        if not path.exists():
            print(f"SKIP (not found): {rel}")
            continue
        n = normalize_svg(path, tiers, dry_run=args.dry_run)
        status = "would change" if args.dry_run else "changed"
        print(f"{rel}: {status} {n} font-size values")


if __name__ == "__main__":
    main()
```
